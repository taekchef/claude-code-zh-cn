const http=require('http'),fs=require('fs'),p=require('path');const root=__dirname;
const s=http.createServer((req,res)=>{let data='';req.on('data',x=>data+=x);req.on('end',()=>{let b={};try{b=JSON.parse(data)}catch{};fs.appendFileSync(p.join(root,'requests.log'),JSON.stringify({url:req.url,stream:b.stream,tools:(b.tools||[]).map(t=>t.name),last:b.messages?.at(-1)?.role})+'\n');if(req.url.includes('count_tokens')){res.end(JSON.stringify({input_tokens:100}));return;}const msg={id:'msg_local_probe',type:'message',role:'assistant',model:b.model||'claude-sonnet-4-6',content:[{type:'tool_use',id:'toolu_local_probe',name:'Skill',input:{skill:'localization-probe'}}],stop_reason:'tool_use',stop_sequence:null,usage:{input_tokens:100,output_tokens:20}};if(!b.stream){res.setHeader('Content-Type','application/json');res.end(JSON.stringify({...msg,content:[{type:'text',text:'local probe'}],stop_reason:'end_turn'}));return;}res.setHeader('Content-Type','text/event-stream');const e=(event,data)=>res.write('event: '+event+'\ndata: '+JSON.stringify(data)+'\n\n');e('message_start',{type:'message_start',message:{...msg,content:[],stop_reason:null}});e('content_block_start',{type:'content_block_start',index:0,content_block:{type:'tool_use',id:msg.content[0].id,name:'Skill',input:{}}});e('content_block_delta',{type:'content_block_delta',index:0,delta:{type:'input_json_delta',partial_json:'{"skill":"localization-probe"}'}});e('content_block_stop',{type:'content_block_stop',index:0});e('message_delta',{type:'message_delta',delta:{stop_reason:'tool_use',stop_sequence:null},usage:{output_tokens:20}});e('message_stop',{type:'message_stop'});res.end();});});s.listen(0,'127.0.0.1',()=>{fs.writeFileSync(p.join(root,'port'),String(s.address().port));console.log('Local UI probe listening',s.address().port)});
